    // Calculate Balances per Bucket
    function calculateStats() {
      let totalIncome = 0;
      let totalExpense = 0;
      const bucketStats = {};

      buckets.forEach(b => {
        bucketStats[b.id] = { allocated: 0, spent: 0, balance: 0 };
      });

      transactions.forEach(tx => {
        if (tx.type === 'income') {
          totalIncome += tx.amount;
          if (tx.allocations) {
            Object.entries(tx.allocations).forEach(([bId, amt]) => {
              if (!bucketStats[bId]) {
                bucketStats[bId] = { allocated: 0, spent: 0, balance: 0 };
              }
              bucketStats[bId].allocated += amt;
              bucketStats[bId].balance += amt;
            });
          }
        } else if (tx.type === 'expense') {
          totalExpense += tx.amount;
          if (!bucketStats[tx.bucketId]) {
            bucketStats[tx.bucketId] = { allocated: 0, spent: 0, balance: 0 };
          }
          bucketStats[tx.bucketId].spent += tx.amount;
          bucketStats[tx.bucketId].balance -= tx.amount;
        }
      });

      return {
        totalIncome,
        totalExpense,
        totalBalance: totalIncome - totalExpense,
        bucketStats
      };
    }

    function handleTransactionSubmit(e) {
      e.preventDefault();
      const amount = parseFloat(document.getElementById('amountInput').value);
      const note = document.getElementById('noteInput').value.trim();
      const date = document.getElementById('dateInput').value;

      if (!amount || amount <= 0 || !note || !date) return;

      if (currentFormMode === 'income') {
        const target = document.getElementById('incomeTargetSelect').value;
        const allocations = {};

        if (target === 'auto') {
          let allocatedSum = 0;
          buckets.forEach((b, idx) => {
            let share = Math.round((amount * b.percentage) / 100);
            if (idx === buckets.length - 1) {
              share = amount - allocatedSum;
            } else {
              allocatedSum += share;
            }
            allocations[b.id] = share;
          });
        } else {
          allocations[target] = amount;
        }

        transactions.unshift({
          id: 'tx_' + Date.now(),
          type: 'income',
          amount,
          note,
          date,
          target,
          allocations
        });

        showToast('ဝင်ငွေမှတ်တမ်းတင်ပြီး ခွဲဝေပေးပြီးပါပြီ');
      } else {
        const bucketId = document.getElementById('expenseBucketSelect').value;
        transactions.unshift({
          id: 'tx_' + Date.now(),
          type: 'expense',
          amount,
          note,
          date,
          bucketId
        });

        showToast('အသုံးစရိတ် မှတ်တမ်းတင်ပြီးပါပြီ');
      }

      saveData();
      document.getElementById('amountInput').value = '';
      document.getElementById('noteInput').value = '';
      closeEntryModal();
      renderAll();
    }

    function deleteTransaction(id) {
      transactions = transactions.filter(tx => tx.id !== id);
      saveData();
      renderAll();
      showToast('မှတ်တမ်းကို ဖျက်လိုက်ပါပြီ');
    }

    function setFilter(filter) {
      currentFilter = filter;
      ['all', 'income', 'expense'].forEach(f => {
        const btn = document.getElementById(`filter-${f}`);
        if (f === filter) {
          btn.className = 'px-3 py-1.5 text-xs font-semibold rounded-lg bg-white dark:bg-slate-900 text-slate-800 dark:text-white shadow-sm transition cursor-pointer';
        } else {
          btn.className = 'px-3 py-1.5 text-xs font-semibold rounded-lg text-slate-600 dark:text-slate-400 hover:text-slate-900 dark:hover:text-white transition cursor-pointer';
        }
      });
      renderTransactions();
      lucide.createIcons();
    }

    function renderTransactions() {
      const list = document.getElementById('transactionList');
      const searchEl = document.getElementById('historySearchInput');
      const query = searchEl ? searchEl.value.toLowerCase().trim() : '';
      list.innerHTML = '';

      const filtered = transactions.filter(tx => {
        const matchesType = currentFilter === 'all' || tx.type === currentFilter;
        const matchesSearch = !query || tx.note.toLowerCase().includes(query);
        return matchesType && matchesSearch;
      });

      if (filtered.length === 0) {
        list.innerHTML = `
          <div class="text-center py-12 px-4">
            <div class="w-12 h-12 rounded-2xl bg-slate-100 dark:bg-slate-800 text-slate-400 flex items-center justify-center mx-auto mb-3">
              <i data-lucide="file-text" class="w-6 h-6"></i>
            </div>
            <p class="text-sm font-semibold text-slate-700 dark:text-slate-200">မှတ်တမ်းများ မရှိသေးပါ</p>
            <p class="text-xs text-slate-400 dark:text-slate-500 mt-1">အပေါ်ရှိ "+ စာရင်းသွင်း & တွက်ချက်" ခလုတ်ကို နှိပ်၍ စတင်ထည့်သွင်းပါ</p>
          </div>
        `;
        lucide.createIcons();
        return;
      }

      filtered.forEach(tx => {
        const isIncome = tx.type === 'income';
        let subtitle = '';

        if (isIncome) {
          if (tx.target === 'auto') {
            subtitle = 'အလိုအလျောက် ရာခိုင်နှုန်းခွဲဝေမှု';
          } else {
            const b = buckets.find(item => item.id === tx.target);
            subtitle = b ? `တိုက်ရိုက်ထည့်သွင်း: ${b.name}` : 'တိုက်ရိုက်ထည့်သွင်းငွေ';
          }
        } else {
          const b = buckets.find(item => item.id === tx.bucketId);
          subtitle = b ? `ကဏ္ဍ: ${b.name}` : 'အသုံးစရိတ်';
        }

        const item = document.createElement('div');
        item.className = 'flex items-center justify-between gap-3 p-3.5 rounded-xl bg-slate-50/90 dark:bg-slate-800/70 hover:bg-slate-100/90 dark:hover:bg-slate-800 border border-slate-200/70 dark:border-slate-700/70 transition group';
        item.innerHTML = `
          <div class="flex items-center gap-3 min-w-0">
            <div class="w-10 h-10 rounded-xl ${isIncome ? 'bg-emerald-100/80 dark:bg-emerald-950 text-emerald-600 dark:text-emerald-400' : 'bg-rose-100/80 dark:bg-rose-950 text-rose-600 dark:text-rose-400'} flex items-center justify-center shrink-0">
              <i data-lucide="${isIncome ? 'arrow-down-left' : 'arrow-up-right'}" class="w-5 h-5"></i>
            </div>
            <div class="min-w-0">
              <p class="text-sm font-bold text-slate-800 dark:text-slate-100 truncate">${tx.note}</p>
              <div class="flex items-center gap-2 text-[11px] text-slate-500 dark:text-slate-400 mt-0.5">
                <span>${tx.date}</span>
                <span>•</span>
                <span class="truncate">${subtitle}</span>
              </div>
            </div>
          </div>

          <div class="flex items-center gap-2.5 shrink-0">
            <span class="text-sm font-bold ${isIncome ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400'}">
              ${isIncome ? '+' : '-'} ${formatMMK(tx.amount)}
            </span>
            <button type="button" onclick="deleteTransaction('${tx.id}')" class="p-1.5 text-slate-400 hover:text-rose-600 dark:hover:text-rose-400 hover:bg-rose-50 dark:hover:bg-rose-950/60 rounded-lg opacity-80 group-hover:opacity-100 transition cursor-pointer" title="ဖျက်ရန်">
              <i data-lucide="trash-2" class="w-4 h-4"></i>
            </button>
          </div>
        `;
        list.appendChild(item);
      });
      lucide.createIcons();
    }

    // Settings Modal Logic
    function openSettingsModal() {
      tempSettingsBuckets = JSON.parse(JSON.stringify(buckets));
      renderSettingsRows();
      document.getElementById('settingsModal').classList.remove('hidden');
      lucide.createIcons();
    }

    function closeSettingsModal() {
      document.getElementById('settingsModal').classList.add('hidden');
    }

    function renderSettingsRows() {
      const container = document.getElementById('settingsBucketsList');
      container.innerHTML = '';

      tempSettingsBuckets.forEach((b, index) => {
        const row = document.createElement('div');
        row.className = 'flex items-center gap-2 p-2.5 rounded-xl bg-slate-50 dark:bg-slate-800/70 border border-slate-200 dark:border-slate-700';
        row.innerHTML = `
          <input type="text" value="${b.name}" onchange="updateTempBucket(${index}, 'name', this.value)" placeholder="ကဏ္ဍအမည်"
            class="flex-1 min-w-0 px-3 py-1.5 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded-lg text-xs font-semibold text-slate-800 dark:text-slate-100 focus:outline-none focus:ring-2 focus:ring-emerald-500" />
          <div class="relative w-24 shrink-0">
            <input type="number" min="0" max="100" value="${b.percentage}" oninput="updateTempBucket(${index}, 'percentage', this.value)"
              class="w-full pl-2.5 pr-6 py-1.5 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded-lg text-xs font-bold text-slate-900 dark:text-white text-right focus:outline-none focus:ring-2 focus:ring-emerald-500" />
            <span class="absolute right-2 top-1/2 -translate-y-1/2 text-xs text-slate-400">%</span>
          </div>
          ${tempSettingsBuckets.length > 1 ? `
            <button type="button" onclick="removeTempBucket(${index})" class="p-1.5 text-slate-400 hover:text-rose-600 dark:hover:text-rose-400 rounded-lg cursor-pointer">
              <i data-lucide="trash-2" class="w-4 h-4"></i>
            </button>
          ` : ''}
        `;
        container.appendChild(row);
      });

      updateSettingsTotal();
      lucide.createIcons();
    }

    function updateTempBucket(index, field, value) {
      if (field === 'percentage') {
        tempSettingsBuckets[index].percentage = parseFloat(value) || 0;
        updateSettingsTotal();
      } else {
        tempSettingsBuckets[index][field] = value.trim();
      }
    }

    function addNewBucketRow() {
      const nextColor = AVAILABLE_COLORS[tempSettingsBuckets.length % AVAILABLE_COLORS.length];
      const nextIcon = AVAILABLE_ICONS[tempSettingsBuckets.length % AVAILABLE_ICONS.length];
      tempSettingsBuckets.push({
        id: 'bucket_' + Date.now(),
        name: 'ကဏ္ဍအသစ်',
        percentage: 0,
        color: nextColor,
        icon: nextIcon
      });
      renderSettingsRows();
    }

    function removeTempBucket(index) {
      tempSettingsBuckets.splice(index, 1);
      renderSettingsRows();
    }

    function updateSettingsTotal() {
      const total = tempSettingsBuckets.reduce((sum, b) => sum + (parseFloat(b.percentage) || 0), 0);
      const totalEl = document.getElementById('settingsTotalPercent');
      const saveBtn = document.getElementById('saveSettingsBtn');

      totalEl.textContent = `${total}%`;
      if (Math.abs(total - 100) < 0.01) {
        totalEl.className = 'text-sm font-bold text-emerald-600 dark:text-emerald-400';
        saveBtn.disabled = false;
        saveBtn.classList.remove('opacity-50', 'cursor-not-allowed');
      } else {
        totalEl.className = 'text-sm font-bold text-rose-600 dark:text-rose-400';
        saveBtn.disabled = true;
        saveBtn.classList.add('opacity-50', 'cursor-not-allowed');
      }
    }

    function saveSettings() {
      const total = tempSettingsBuckets.reduce((sum, b) => sum + (parseFloat(b.percentage) || 0), 0);
      if (Math.abs(total - 100) >= 0.01) return;

      buckets = tempSettingsBuckets.map(b => ({
        ...b,
        name: b.name || 'အမည်မသိကဏ္ဍ'
      }));
      saveData();
      closeSettingsModal();
      renderAll();
      showToast('ရာခိုင်နှုန်း သတ်မှတ်ချက်များကို သိမ်းဆည်းပြီးပါပြီ');
    }

    function executeResetAllData() {
      localStorage.removeItem('mm_budget_buckets');
      localStorage.removeItem('mm_budget_transactions');
      buckets = JSON.parse(JSON.stringify(DEFAULT_BUCKETS));
      transactions = [];
      closeResetConfirmModal();
      closeSettingsModal();
      renderAll();
      showToast('စာရင်းအားလုံးကို ပြန်လည်ရှင်းလင်းပြီးပါပြီ');
    }

    // Export CSV
    function exportToCSV() {
      if (transactions.length === 0) {
        showToast('ထုတ်ယူရန် မှတ်တမ်းမရှိသေးပါ');
        return;
      }

      const headers = ['Date', 'Type', 'Note', 'Amount (MMK)', 'Bucket / Details'];
      const rows = transactions.map(tx => {
        const typeLabel = tx.type === 'income' ? 'Income' : 'Expense';
        let detail = '';
        if (tx.type === 'income') {
          detail = tx.target === 'auto' ? 'Auto Allocated' : (buckets.find(b => b.id === tx.target)?.name || '');
        } else {
          detail = buckets.find(b => b.id === tx.bucketId)?.name || '';
        }
        return [tx.date, typeLabel, `"${tx.note.replace(/"/g, '""')}"`, tx.amount, `"${detail}"`].join(',');
      });

      const csvContent = '\uFEFF' + [headers.join(','), ...rows].join('\n');
      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = `budget_records_${new Date().toISOString().slice(0, 10)}.csv`;
      link.click();
      URL.revokeObjectURL(url);
      showToast('CSV ဖိုင်အဖြစ် ဒေါင်းလုဒ်ဆွဲပြီးပါပြီ');
    }

    // Toast Helper
    let toastTimeout;
    function showToast(message) {
      const toast = document.getElementById('toast');
      document.getElementById('toastMessage').textContent = message;
      toast.classList.remove('translate-y-24', 'opacity-0');
      clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => {
        toast.classList.add('translate-y-24', 'opacity-0');
      }, 3000);
    }
